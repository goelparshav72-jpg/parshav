#include <stdio.h>

int main()
{
    int n;
    int nums[100];
    int answer[100];

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for(int i = 0; i < n; i++)
    {
        scanf("%d", &nums[i]);
    }

    // Calculate product except self
    for(int i = 0; i < n; i++)
    {
        answer[i] = 1;

        for(int j = 0; j < n; j++)
        {
            if(i != j)
            {
                answer[i] = answer[i] * nums[j];
            }
        }
    }

    printf("[");

    for(int i = 0; i < n; i++)
    {
        printf("%d", answer[i]);

        if(i != n - 1)
        {
            printf(",");
        }
    }

    printf("]");

    return 0;
}
